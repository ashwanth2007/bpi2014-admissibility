% IEEEtran sets a description label inside a fixed-width box, so a label wider than
% that box prints on top of the item text instead of pushing it along. The three
% operator names in the multi-objective assignment section are wider than the box,
% which overprinted their definitions. Restore the standard LaTeX run-in behaviour,
% where a long label simply displaces the text that follows it.
\providecommand*\descriptionlabel[1]{}
\renewcommand*\descriptionlabel[1]{\hspace\labelsep\normalfont\bfseries #1}
\makeatletter
\renewenvironment{description}
  {\list{}{\labelwidth\z@ \itemindent-\leftmargin
           \listparindent\parindent
           \let\makelabel\descriptionlabel}}
  {\endlist}
\makeatother

% redefined to their starred forms rather than editing 39 floats in the shared sources.

\renewenvironment{table}[1][]{\@dblfloat{table}[tp]}{\end@dblfloat}
\renewenvironment{figure}[1][]{\@dblfloat{figure}[tp]}{\end@dblfloat}
\makeatother
